HyperParallel 模型适配流程 S0～S6
================================

整理依据

- 当前代码版本为 f8572d71，refactor(agent): isolate the RSI workflow under .agent。
- 总入口为 .agent/skills/hf-model-integration/SKILL.md。
- 该文件 Bring-up stage machine 表是阶段划分、交接产物和退出条件的依据。
- 下文的普通函数名、对象字段和调用顺序是便于阅读的伪代码组织。
- 每阶段先列阶段表直接要求，再给伪代码、引用展开及对应文件。
- 所有路径相对仓库根目录。代码块是流程说明，执行时以实际脚本参数为准。

动作约定

普通函数(输入) -> Agent 完成分析、编写、配置、检查等任务，返回结果。
SKILL(名称, 输入) -> 按指定 Skill 执行，返回其产物。
READ(路径, sections=章节) -> 阅读参考资料、模板或已有产物。
SCRIPT(路径, 参数) -> 执行脚本，取得退出码、输出及产物路径。
ASK(问题, 选项) -> 等待用户决定。
WRITE(路径, 内容) -> 持久化内容，返回文件路径。
SPAWN(任务, 输入) / WAIT(句柄) -> 委派任务及等待结果。

SPAWN/WAIT 是描述可选执行组织的符号。当前阶段表没有要求各阶段必须使用子 Agent。
if、for、while 表达控制流程。检查函数可返回缺失项列表，空列表表示通过。
return 将结果交给调用方，WRITE 将产物写入存储，两者可以使用同一份内容。
函数返回的逻辑对象可以包含实际文件路径。对象字段不代表脚本原本就返回同名 JSON。
SKILL 的输入表示任务和范围，具体执行过程需继续遵循该 Skill 的正文。
每阶段持久化产物的实际路径由运行配置确定，这里不规定额外的中文文件名。

主流程

fun 模型适配(任务, 运行目录) {
    // 主 Skill 要求按顺序推进，阶段产物作为下一阶段的接口。
    // 恢复时读取运行目录，定位第一个缺失的阶段产物，并核对当前资产版本。
    S0结果 = S0_调研规划(任务)
    if S0结果.状态 != "PASS" { return S0结果 }
    WRITE(任务.阶段产物路径.S0, S0结果)

    S1结果 = S1_适配实现(任务, S0结果.迁移规格)
    if S1结果.状态 != "PASS" { return S1结果 }
    WRITE(任务.阶段产物路径.S1, S1结果)

    S2结果 = S2_单节点正确性(任务, S0结果, S1结果)
    if S2结果.状态 != "PASS" { return S2结果 }
    WRITE(任务.阶段产物路径.S2, S2结果)

    S3结果 = S3_分布式等价(任务, S0结果, S1结果, S2结果)
    if S3结果.状态 != "PASS" { return S3结果 }
    WRITE(任务.阶段产物路径.S3, S3结果)

    S4结果 = S4_分析与优化(任务, S1结果, S3结果)
    if S4结果.状态 != "PASS" { return S4结果 }
    WRITE(任务.阶段产物路径.S4, S4结果)

    S5结果 = S5_验收集成(任务, S0结果, S1结果, S3结果, S4结果)
    WRITE(任务.阶段产物路径.S5, S5结果)
    if S5结果.状态 != "PASS" { return S5结果 }

    if 任务.要求形式验证 {
        S6结果 = S6_形式验证(任务, S5结果)
        WRITE(任务.阶段产物路径.S6, S6结果)
        return { 验收交付: S5结果, 形式验证: S6结果 }
    }
    return S5结果
}

S0～S5 按阶段推进，S6 是原阶段表中的可选形式验证阶段。
WRITE 记录的是已执行步骤的产物及其引用，不会将未运行步骤标为通过。


S0 调研规划
============

阶段表直接要求

- 使用本 Skill 的 bringup-field-checklist.md §1～§3，以及 parallel-strategy-analyzer。
- 产物为迁移规格，包含结构表、硬件与依赖概况、并行计划、风险清单。
- 硬件与依赖概况使用 templates/environment-profile.md。
- 退出条件为规格完整，每项无先例语义均已标记待上报。

fun S0_调研规划(任务) {
    清单 = READ(
        ".agent/skills/hf-model-integration/references/bringup-field-checklist.md",
        sections=[1, 2, 3]
    )
    环境模板 = READ(
        ".agent/skills/hf-model-integration/templates/environment-profile.md"
    )

    // 以下动作展开自清单 §1～§3。
    模型来源 = 阅读并固定模型源码和依赖版本(任务, 清单)
    权重结构 = 推导发布权重名称和结构(模型来源, 清单)
    语义差异 = 识别通用模块未覆盖的模型语义(模型来源, 清单)
    硬件信息 = 只读调查硬件并记录探测来源(任务.目标环境, 清单)
    已有权重 = 清点已有权重并核验完整性(任务.允许搜索位置, 清单)
    依赖概况 = 调查依赖版本和覆盖需求(模型来源, 任务.目标环境, 清单)
    结构表 = 整理模型结构表(模型来源, 权重结构, 语义差异)
    环境概况 = 填写模板(环境模板, 硬件信息, 依赖概况)

    并行计划 = SKILL(
        "parallel-strategy-analyzer",
        { 结构表, 环境概况, 任务.并行要求, 任务.序列长度, 任务.全局批量 }
    )

    风险清单 = 汇总风险(语义差异, 已有权重, 依赖概况, 并行计划)
    待上报语义 = 标记无先例语义(语义差异)
    迁移规格 = {
        模型来源, 结构表, 环境概况, 已有权重,
        并行计划, 风险清单, 待上报语义
    }

    缺失项 = 检查规格完整性及无先例语义标记(迁移规格)
    return {
        状态: if 缺失项.为空 "PASS" else "BLOCKED",
        迁移规格, 缺失项
    }
}

S1 适配实现
============

阶段表直接要求

- 使用本 Skill、add-unit-test、feature-doc†。
- 产物为通用工具包上的 recipe，需要 provider 才提供适配器。
- 发布布局检查使用清单 §5 的脚本，退出码为 0，并声明 full-shape 或 names-only 覆盖。
- 退出条件为布局检查通过，适用的家族测试通过。

fun S1_适配实现(任务, 迁移规格) {
    // 关联参考来自主 Skill 的 Load 要求和清单 §4。
    // S1 行直接指定的清单章节是 §5。
    构建说明 = READ(
        ".agent/skills/hf-model-integration/references/bringup-field-checklist.md",
        sections=[4]
    )
    布局说明 = READ(
        ".agent/skills/hf-model-integration/references/bringup-field-checklist.md",
        sections=[5]
    )

    工具包配置 = 配置通用模型构建和训练Recipe(迁移规格, 构建说明) # 按照s0并行计划中的推荐并行配置，生成recipe
    需要Provider = 判断任务所需能力是否需要模型专属扩展(迁移规格, 工具包配置)
    适配器 = None

    if 需要Provider {
        分析指南 = READ(
            ".agent/skills/hf-model-integration/references/model-analysis-and-adapter.md"
        )
        适配器 = 实现所需的模型适配器(迁移规格, 分析指南)
    }

    测试 = SKILL("add-unit-test", { 迁移规格, 工具包配置, 适配器 })
    文档 = SKILL("feature-doc", { 迁移规格, 工具包配置, 适配器 })

    布局检查 = SCRIPT(
        ".agent/skills/hf-model-integration/scripts/check_released_layout.py",
        ["--model-dir", 任务.模型目录]
    )
    覆盖说明 = 从检查输出记录完整形状或仅名称覆盖(布局检查)
    家族测试结果 = 运行适用的家族测试(测试, 工具包配置, 适配器)

    return {
        状态: if 布局检查.退出码 == 0 && 家族测试结果.通过 "PASS" else "BLOCKED",
        工具包配置, 适配器, 测试, 文档, 布局检查, 覆盖说明, 家族测试结果
    }
}

Provider 的含义和条件

- 当前实现位于 hyper_parallel/models/adapter_spec.py 的 ModelAdapterSpec。
- 通用构建与规划已覆盖需求时可以走 HF 原生零注册路线。
- 模型专属分片规则对应 sharding_rules。
- 模型专属模块替换与权重转换对应 replacements。
- 模型专属序列或专家通信对应 context_parallel、expert_parallel。
- 特殊 FSDP 包装、排除和执行顺序对应 fsdp_wrap_modules、fsdp_excluded_subtrees、fsdp_execution_order。
- 专属重计算和 loss 适配对应 recompute、loss。
- 启用某个并行维度时，要检查该模型语义是否已被通用实现或 recipe 覆盖。

S2 单节点正确性
================

阶段表直接要求

- 使用清单 §6、hf-model-precision-validation。
- 与普通 HF 构建做同进程逐位比较，保存报告，并保存训练冒烟 loss 曲线。
- 零注册路线用该报告替代 integration_handoff.yaml，作为进入 S3 的前置证据。
- 退出条件为前向一致，loss 有限且下降。

fun S2_单节点正确性(任务, S0结果, S1结果) {
    首轮检查 = READ(
        ".agent/skills/hf-model-integration/references/bringup-field-checklist.md",
        sections=[6]
    )
    真实权重冒烟 = 检查真实权重加载前向和生成(任务, S1结果, 首轮检查) # 比较权重和一次前向

    if 任务.模型属于通用比较脚本支持范围 {
        同进程比较 = SCRIPT(
            ".agent/examples/hf_text_bringup/reproduce/parity_check.py",
            任务.同进程比较参数
        )
        // 脚本对比参数、缓冲区和 logits，报告含每项差异。
        // 参数中指定 --model-dir、--dtype、--output 等实际支持的选项。
    } else {
        同进程比较 = 按清单执行并持久化真实构建路径的逐位比较(
            任务, S0结果.迁移规格.模型来源, S1结果
        )
        // 通用工具包以外的模型需选择适用的参考和比较入口，记录覆盖范围。
    }

    模块验证 = None
    if S1结果.适配器 != None { # TODO 比较单个替换模块前向和反向，这个感觉顺序有问题，应该放在开头
        模块验证 = SKILL(
            "hf-model-precision-validation",
            { 范围: "模块一致性", 参考实现: S0结果.迁移规格.模型来源, 候选: S1结果 }
        )
        交接证据 = 模块验证.integration_handoff及引用报告
    } else {
        交接证据 = 同进程比较.报告路径
    }

    冒烟结果 = 运行单节点训练冒烟(S1结果.工具包配置, 首轮检查)
    缺失项 = 检查前向一致及Loss有限且下降( # TODO 比较实际训练，好像没有规定反向要求
        同进程比较, 模块验证, 冒烟结果
    )

    return {
        状态: if 缺失项.为空 "PASS" else "BLOCKED",
        真实权重冒烟, 同进程比较, 模块验证, 交接证据, 冒烟结果, 缺失项
    }
}

S3 分布式等价
==============

阶段表直接要求

- 使用 hf-model-precision-validation、cluster-debug†。
- 开启 model_integration.mode=runtime，使用全精度 loss、梯度范数及逐步全局输入哈希。
- 重计算开关的比较、确定性重跑要求精确一致。
- 跨布局从相同发布权重和 token-uniform 语料开始，早期步精确一致，记录分叉起点及增长。
- 通用工具包 judge_equivalence.py 逐步委托 compare_scalar_curves 判定。
- 退出条件为所有目标布局满足各自标准。

fun S3_分布式等价(任务, S0结果, S1结果, S2结果) {
    精度指南 = READ(".agent/skills/hf-model-precision-validation/SKILL.md")
    交接检查 = 核验S2交接证据及引用文件(S2结果.交接证据)
    if !交接检查.通过 {
        return { 状态: "BLOCKED", 缺失项: 交接检查.缺失项 }
    }

    # TODO 有必要进行这一步吗？rsi优化出最佳配置，有必要使用其他配置吗？另外是不是不需要预先生成一个检查矩阵，再后续rsi迭代过程中，再进行正确性检查不行吗？
    矩阵 = 制定目标布局重计算和重跑矩阵(S0结果.迁移规格, 精度指南) # 围绕设置并行，验证一下不同并行和重计算组合的精度
    比较条件 = 固定发布权重输入及运行证据设置(矩阵, 任务, 精度指南)

    矩阵结果 = SKILL(
        "hf-model-precision-validation",
        { 范围: "分布式等价", 候选: S1结果, S2交接: S2结果, 矩阵, 比较条件 }
    )

    if 矩阵结果.存在分布式故障 {
        诊断结果 = SKILL("cluster-debug", { 矩阵, 失败证据: 矩阵结果 })
        return { 状态: "BLOCKED", 矩阵, 矩阵结果, 诊断结果 }
        // 修复后的重跑需产生新证据，再检查受影响单元。
    }

    缺失项 = 检查所有目标布局及其比较证据(矩阵, 矩阵结果)
    return {
        状态: if 缺失项.为空 "PASS" else "BLOCKED",
        矩阵, 比较条件, 矩阵结果, 缺失项
    }
}

S4 分析与优化
==============

阶段表直接要求

- 使用 profiling†、memory-analysis†、训练 RSI 协议、autoresearch†。
- 产物为通过检查的性能基线、瓶颈清单和候选实验凭据。

        可审阅变更集: 汇总提交差异及测试证据(待验收版本)

parallel-strategy-analyzer：并行策略规划

fun 并行策略规划(任务) {
    READ(".agent/skills/parallel-strategy-analyzer/references/io-contract.md")
    信息 = 按01收集模型设备及约束(任务)
    // 模型名称或参数量、设备数量、设备类型为必要信息。
    // 文档未提供时的默认值：seq_len=4096、global_batch=1024、micro_batch=1、bf16、adam。
    // 使用已有实际配置时保留其工作负载，不把估算默认值当成任务事实。
    全局基线 = 按02估算无并行显存及FLOPs(信息)
    候选 = 按03枚举并剪枝并行组合(信息, 全局基线)

    for 方案 in 候选 {
        通信与气泡 = 按04估算各维通信及重叠(方案, 信息)
        分片显存 = 按05估算各设备状态和激活占用(方案, 信息)
        评分 = 按05文档公式排序(方案, 分片显存, 通信与气泡)
        // 超过90%显存容量的候选淘汰。
        // 评分 = 100*估算MFU + 20*(1-abs(显存利用率-0.75)) - 2*启用并行维度数。
        // 估算MFU来自理论计算时间、暴露通信和气泡，不是实测MFU。
        保存候选分析(方案, 评分)
    }

    return 汇总基线推荐及备选(
        推荐布局, DeviceMesh示例, 分片显存, 通信分解,
        气泡, 显存优化建议, Top3备选, 估算假设及限制
    )
}

- 实验控制器运行于Linux，要求可复现的模型状态、数据、启动命令、设备、目标和预算。
- 不自带集群专属SSH或torchrun启动器；首次接入需将实际训练入口连接到命令协议。

autoresearch：提交、验证、测量与保留恢复
=======================================

职责和边界

- 入口为 .agent/skills/autoresearch/SKILL.md。
- 适合指定目标文件的算子、图或其他代码优化实验，也可接入结构化训练协议。
- 可用于 S4；S5调用finalize检查经验复盘是否完整。
- Agent选择想法、修改目标并维护解释性记录；工具执行确定性的实验协议。
- 不等同于文件开头引用的外部karpathy/autoresearch实现，本节描述本仓库工具。

fun 通用优化实验(任务) {
    SCRIPT(".agent/skills/autoresearch/scripts/autoresearch.py", ["init", ...])
    填写ManualSetup并核对RunJson()
    // 明确目标文件、benchmark/gate命令、指标、噪声、预算和RSI模式。
    // 命令为argv列表；训练协议要求独立gate和benchmark及结构化凭据。
    SCRIPT(".agent/skills/autoresearch/scripts/autoresearch.py", ["baseline", ...])

    while Agent判断目标未达到且预算尚有剩余 {
        读取StatusIdeasLearnings和ExperimentLog()
        优先审阅已满足重试条件的候选()
        修改声明的目标文件()
        结果 = SCRIPT(".agent/skills/autoresearch/scripts/autoresearch.py", ["iterate", ...])
        // loop.iterate()拒绝目标文件和运行目录以外的脏改动。
        // 先提交候选，再执行门禁与测量，确保测量对应已提交代码。
        // 缺少必需生效标记或门禁失败为crash，不让漂亮指标掩盖失败。
        // 按指标方向、噪声和可选重复确认决定keep/discard/crash。
        // 保存结果与候选凭据；非keep用新的正向提交恢复实验前目标文件。
        更新ExperimentLogIdeas和Learnings(结果)
        // CLI在baseline和iterate后调用_learning_checkpoint()。
        // acceptance模式默认在验收处理；continuous模式支持逐轮处理。
    }

    RSI结果 = SCRIPT(".agent/skills/autoresearch/scripts/autoresearch.py", ["finalize", ...])
    return 汇总实际最佳结果及完整RSI报告()
}
